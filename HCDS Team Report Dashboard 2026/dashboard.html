-- 1. Create Team Tasks Table
CREATE TABLE IF NOT EXISTS team_tasks (
    id TEXT PRIMARY KEY,
    task_date TEXT NOT NULL,
    team_member TEXT NOT NULL,
    task_title TEXT NOT NULL,
    description TEXT,
    product TEXT,
    priority TEXT DEFAULT 'Medium',
    status TEXT NOT NULL,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);

-- 2. Enable RLS and grant public access for team
ALTER TABLE team_tasks ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow team read write" ON team_tasks FOR ALL USING (true) WITH CHECK (true);

-- 3. Enable Realtime WebSocket Broadcasting
ALTER PUBLICATION supabase_realtime ADD TABLE team_tasks;

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Activity Log") || SpreadsheetApp.getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var tasks = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (row[0]) {
      tasks.push({
        id: String(row[0]),
        date: String(row[1] || ''),
        member: String(row[2] || 'Unassigned'),
        title: String(row[3] || ''),
        desc: String(row[4] || ''),
        product: String(row[5] || 'General'),
        priority: String(row[6] || 'Medium'),
        status: String(row[7] || 'Pending')
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", tasks: tasks }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Activity Log") || SpreadsheetApp.getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var action = params.action; // 'upsert', 'delete', 'syncAll'
  
  if (action === 'upsert') {
    var t = params.task;
    var foundIndex = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(t.id)) { foundIndex = i + 1; break; }
    }
    var rowData = [t.id, t.date, t.member, t.title, t.desc || '', t.product || 'General', t.priority || 'Medium', t.status || 'Pending'];
    if (foundIndex > 0) {
      sheet.getRange(foundIndex, 1, 1, 8).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", id: t.id }))
      .setMimeType(ContentService.MimeType.JSON);
  } else if (action === 'delete') {
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(params.id)) { sheet.deleteRow(i + 1); break; }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } else if (action === 'syncAll') {
    sheet.clearContents();
    sheet.appendRow(["Task ID", "Date", "Team Member", "Task Title", "Description", "Product", "Priority", "Status"]);
    var rows = params.tasks.map(function(t) {
      return [t.id, t.date, t.member, t.title, t.desc || '', t.product || 'General', t.priority || 'Medium', t.status || 'Pending'];
    });
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, 8).setValues(rows);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
